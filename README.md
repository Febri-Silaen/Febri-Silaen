class DigitalArchitect {
  readonly name: string     = "Febri Silaen";
  readonly role: string     = "Full Stack Developer";
  readonly location: string = "Indonesia 🇮🇩";
  readonly vision: string   = "Mengubah imajinasi menjadi realitas digital";

  private passion: { isAlive: () => boolean } = { isAlive: () => true };

  private mission: string[] = [
    "🚀 Membangun sistem yang cerdas dan intuitif",
    "✨ Menciptakan pengalaman pengguna yang memukau",
    "💡 Berinovasi tanpa henti dalam teknologi",

  ];

  private code():   void { console.log("💻 Writing elegant code..."); }
  private learn():  void { console.log("📚 Absorbing new knowledge..."); }
  private evolve(): void { console.log("🦋 Becoming better every day..."); }
  private impact(): void { console.log("🌟 Making a difference..."); }

  public createFuture(): void {
    while (this.passion.isAlive()) {
      this.code();
      this.learn();
      this.evolve();
      this.impact();
    }
  }
}
